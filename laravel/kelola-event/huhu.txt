Untuk Admin:
	•	Dashboard Admin:
URL: http://localhost:8000/admin/dashboard
Controller: AdminController@dashboard
	•	Buat Event:
URL: POST http://localhost:8000/admin/create-event
Controller: AdminController@createEvent
	•	Kelola Event:
URL: http://localhost:8000/admin/manage-event/{event}
Controller: AdminController@manageEvent
	•	Konfirmasi Kehadiran:
URL: POST http://localhost:8000/admin/confirm-attendance/{registration}
Controller: AdminController@confirmAttendance

Untuk User:
	•	Dashboard User:
URL: http://localhost:8000/user/dashboard
Controller: UserController@dashboard
	•	Daftar Event:
URL: POST http://localhost:8000/user/register/{event}
Controller: UserController@register
	•	Konfirmasi Kehadiran:
URL: POST http://localhost:8000/user/confirm-attendance
Controller: UserController@confirmAttendance
